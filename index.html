Declare a cdal::Feature, resolve permanent IDs with require(), and install hooks through cdal::hooks so mods never clobber each other.

C++Feature guard + shared hook chain (cdal::hooks)
#include <cdal/cdal.hpp>
#include <cdal/hooks.hpp>
#include <cdal/ids.hpp>

using UpdateFn = void (*)(void*);
static cdal::hooks::FunctionHook<UpdateFn> g_hook;
static cdal::Feature feat{"My Mod", "Core hook"};

void on_update(void* self) { g_hook.original()(self); }

void init() {
    std::string missing, err;
    if (!feat.require({cdal::ids::Game::Update}, &missing)) return;
    if (!g_hook.install(cdal::ids::Game::Update, &on_update, {"My Mod", 0}, &err)) feat.off(err);
}
